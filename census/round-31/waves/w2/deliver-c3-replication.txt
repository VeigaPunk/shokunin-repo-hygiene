  t=$(G rev-parse "$r")

done
G gc --prune=now
printf '### object present after dropping the descendant keep ref and pruning\n'
G cat-file -e "$B" 2>/dev/null
printf 'present_rc=%s\n' "$?"

# =====================================================================================
printf '### done\n'
exit 0
