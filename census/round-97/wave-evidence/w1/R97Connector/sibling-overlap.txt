def reap_fifo_orphans(fifo):
    """SIGKILL any survivor still carrying this fifo's path on its cmdline.

    The sox ffmpeg bridge grandchild can escape the child's process group
    (r70-critic observed one alive 5+ min after a blocked ladder leg —
    killpg is necessary but not sufficient); a cmdline sweep is the
    verdict-neutral backstop for both fifo legs."""
    needle = str(fifo).encode()
    for pid_dir in Path("/proc").iterdir():
        if not pid_dir.name.isdigit():
            continue
        try:
            cmd = (pid_dir / "cmdline").read_bytes()
        except OSError:
            continue
        if needle in cmd:
            try:
                os.kill(int(pid_dir.name), signal.SIGKILL)
            except OSError:
                pass

-- audio lifespan probe: /proc/<pid>/cmdline used for DETECTION only (find_procs), not reaping
def find_procs(fifo_str):
    """pid -> (comm, wchan, ppid) for sox/ffmpeg referencing this fifo."""
    out = {}
    for pid in filter(str.isdigit, os.listdir("/proc")):
        try:
            with open(f"/proc/{pid}/cmdline", "rb") as fh:
                cmd = fh.read().decode(errors="replace").replace("\0", " ")
            if fifo_str not in cmd:
                continue
            comm = Path(f"/proc/{pid}/comm").read_text().strip()
            if comm not in ("sox", "ffmpeg"):
                continue
            wchan = Path(f"/proc/{pid}/wchan").read_text().strip()
            ppid = re.search(r"PPid:\s*(\d+)",
                             Path(f"/proc/{pid}/status").read_text())
            out[pid] = (comm, wchan, ppid.group(1) if ppid else "?")
